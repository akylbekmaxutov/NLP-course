import re
from collections import Counter

import numpy as np
from gensim.models import Word2Vec

word2vec = Word2Vec.load("models/word2vec.model")

a = "қосымша өте жақсы емес"      # "the app is not very good"
b = "емес жақсы өте қосымша"      # the same four words, shuffled

def bag_of_words(text):
    return Counter(re.findall(r"\w+", text.lower()))

def document_vector(text):
    words = re.findall(r"\w+", text.lower())
    return np.mean([word2vec.wv[w] for w in words if w in word2vec.wv], axis=0)

print("«" + a + "»")
print("«" + b + "»")

print("same document vector  :", bool(np.allclose(document_vector(a), document_vector(b))))
print("largest difference    :", float(np.abs(document_vector(a) - document_vector(b)).max()),

Теория · Модель

Рекурренттік желі дегеніміз не

Әр сөзге бір рет қолданылатын және оқығанының бәрінің түйінін алып жүретін бір шағын желі.

Рекурренттік нейрондық желі жасырын күй ұстайды: нөлден басталып, әр сөзде қайта жазылатын вектор, h = f(h, сөз). Әр орында сол f және сол салмақтар жұмыс істейді. Соңғы сөзден кейін дәл сол жалғыз вектор — модельдің пікір туралы түйіні.

Төрт рет салынған бір желі, төрт желі емес. Оның салмақтары барлық орындарға ортақ, сондықтан үш сөзді пікір де, қырық сөзді пікір де дәл сол параметрлер арқылы өтеді.

Сөздер ретімен келетіндіктен, жақсы емес пен емес жақсы енді әртүрлі жасырын күй береді. Қосымша механиканың бүкіл себебі осы: курстағы басқа ештеңе бұл екеуін ажырата алмайтын.

4-дәрістен бері не өзгерді

эмбеддингтер
бұдан былай бөлек қадам емес — олар желінің бірінші қабаты болды әрі қалғанымен бірге оқытылады
не оқытылады
салмақтар дистрибутивтік гипотезадан емес, белгілер бойынша градиенттік түсуден келеді
шығыс
сөздерді ретімен көрген бір жасырын күй, содан кейін кластарға сызықтық қабат

Оқылуы: эмбеддинг қадамы жоғалған жоқ. Ол бірінші қабат болды — әрі дәрістің соңында біз оны 4-дәрісте оқытылған векторлардан бастаймыз.

Теория · Шешім

Қарапайым RNN неге ұмытады және LSTM не істейді

Жасырын күй әр сөзде қайта жазылады. Оны қырық рет қайта жазыңыз да, басында ондағы не болса, соның бәрі жоғалады.

Жоғалатын градиент

Бірінші сөздің маңызды болғанын үйрену үшін оқыту сигналы аралықтағы әр қадам арқылы кері өтуі керек, ал әр қадамда ол сол бір салмақ матрицасына көбейтіледі. Бірден кіші санды өзіне қырық рет көбейтіңіз — ол іс жүзінде нөл; бірден үлкенді көбейтсеңіз — жарылады. Іс жүзінде қарапайым RNN соңғы бірнеше сөзді жақсы үйренеді де, ұзын пікірдің басын мүлде дерлік үйренбейді. Оқыту циклындағы clip_grad_norm_ — жарылатын жартысына арналған стандартты жамау; жоғалатын жартысына жамау жоқ.

LSTM екінші векторды — ұяшық күйін — қосады, ол қадамнан қадамға дерлік қолжетпей өтеді: оған қосады және одан өшіреді, бірақ оны қайта жазбайды. Әрқайсысы 0 мен 1 арасында сан беретін шағын желі болатын үш қақпа нені өшіру, нені қосу және нені көрсету керегін шешеді.

Ұяшық күйі жоғарыдан түзу сызықпен өтеді. Қақпалар оны өңдейді, бірақ ауыстырмайды.

Үш қақпа
forget
ұяшық күйін көбейтіп, нені тастауды таңдайды. 1-ге жақын мән жадты көп сөз бойы тірі ұстайды
кіріс
осы қадамда жаңа кандидат мәнінің қаншасын қосуды шешеді
шығыс
ұяшық күйінің қаншасы көрінетін жасырын күйге айналатынын шешеді

Бұл неге көмектеседі: ұяшық күйінің бір қадамнан екіншісіне жолы — матрицаға көбейту емес, қосу, сондықтан градиент нөлге сықпай, әлдеқайда алысқа кері жүре алады.

Бағасы — көлем. RNN бір нәрсе есептейтін жерде LSTM төртеуін есептейді — үш қақпа мен кандидат мәні, — сондықтан оның рекурренттік салмақ матрицалары төрт есе биік. Мұны төмендегі параметрлер тізімінен тікелей көресіз: қарапайым RNN-де (128, 100) болатын жерде (512, 100).

Пікірлердің ұзындығы әртүрлі, ал тензор — тікбұрыш. Біз батчтағы ең ұзын пікірге дейін толтырамыз да, қасында нағыз ұзындықтарды береміз, сонда желі әр пікір шынымен бітетін жерде тоқтай алады.

Практика · Модель

PyTorch-тағы желі

Үш қабат және он бір жол. RNN мен LSTM арасында ауысу — бір сөз. 

import torch
import torch.nn as nn

PAD = 0

class Recurrent(nn.Module):
    """Embedding -> RNN or LSTM -> Linear."""

    def __init__(self, vocab_size, classes, kind="lstm", dim=100, hidden=128, dropout=0.3):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, dim, padding_idx=PAD)
        cell = nn.LSTM if kind == "lstm" else nn.RNN
        self.rnn = cell(dim, hidden, batch_first=True)
        self.dropout = nn.Dropout(dropout)
        self.out = nn.Linear(hidden, classes)

    def forward(self, ids, lengths):
        embedded = self.dropout(self.embedding(ids))          # (batch, words, dim)
        packed = nn.utils.rnn.pack_padded_sequence(
            embedded, lengths, batch_first=True, enforce_sorted=False)
        _, state = self.rnn(packed)
        hidden = state[0] if isinstance(state, tuple) else state   # LSTM returns (h, c)
        return self.out(self.dropout(hidden[-1]))             # (batch, classes)

import os
import re
import time

from dotenv import load_dotenv

load_dotenv(".env")

import pandas as pd
import torch
import torch.nn as nn
from datasets import load_dataset
from sklearn.metrics import classification_report, confusion_matrix
from torch.utils.data import DataLoader

PAD, UNK, SEED = 0, 1, 42

# ---- data ----------------------------------------------------------------
def split(name, per_class):
    df = load_dataset("issai/kazsandra", "score_classification", split=name,
                      token=os.getenv("HF_TOKEN")).to_pandas()
    per_class = min(per_class, df["label"].value_counts().min())   # 2 stars is the rarest
    df = pd.concat([g.sample(per_class, random_state=SEED) for _, g in df.groupby("label")])
    return df.sample(frac=1, random_state=SEED).reset_index(drop=True)

def tokenize(text):
    return re.findall(r"\w+", str(text).lower())

NAMES = ["1 star", "2 stars", "3 stars", "4 stars", "5 stars"]
train_df, test_df = split("train", 4000), split("test", 1000)

counts = {}
for text in train_df["text_cleaned"]:
    for word in tokenize(text):
        counts[word] = counts.get(word, 0) + 1
vocab = {"<pad>": PAD, "<unk>": UNK}
for word in sorted((w for w, c in counts.items() if c >= 2), key=lambda w: (-counts[w], w)):
    vocab[word] = len(vocab)

def encode(df, max_len=60):
    rows = []
    for text, label in zip(df["text_cleaned"], df["label"]):
        ids = [vocab.get(w, UNK) for w in tokenize(text)][:max_len]
        rows.append((ids or [UNK], int(label)))
    return rows

def collate(batch):
    lengths = torch.tensor([len(ids) for ids, _ in batch])
    padded = torch.full((len(batch), int(lengths.max())), PAD, dtype=torch.long)
    for i, (ids, _) in enumerate(batch):
        padded[i, :len(ids)] = torch.tensor(ids)
    return padded, lengths, torch.tensor([y for _, y in batch])

train_rows, test_rows = encode(train_df), encode(test_df)
test_batches = DataLoader(test_rows, 64, shuffle=False, collate_fn=collate)

def shuffled_batches():
    """A fresh generator each time, so every model sees the same batch order."""
    return DataLoader(train_rows, 64, shuffle=True, collate_fn=collate,
                      generator=torch.Generator().manual_seed(SEED))

# ---- model ---------------------------------------------------------------
class Recurrent(nn.Module):
    def __init__(self, vocab_size, classes, kind, dim=100, hidden=128, dropout=0.3):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, dim, padding_idx=PAD)
        self.rnn = (nn.LSTM if kind == "lstm" else nn.RNN)(dim, hidden, batch_first=True)
        self.dropout = nn.Dropout(dropout)
        self.out = nn.Linear(hidden, classes)

    def forward(self, ids, lengths):
        packed = nn.utils.rnn.pack_padded_sequence(
            self.dropout(self.embedding(ids)), lengths, batch_first=True, enforce_sorted=False)
        _, state = self.rnn(packed)
        hidden = state[0] if isinstance(state, tuple) else state
        return self.out(self.dropout(hidden[-1]))

# ---- train ---------------------------------------------------------------
def accuracy(model, batches):
    model.eval()
    right = total = 0
    with torch.no_grad():
        for ids, lengths, labels in batches:
            right += int((model(ids, lengths).argmax(1) == labels).sum())
            total += len(labels)
    return right / total

def predict(model, batches):
    model.eval()
    out, gold = [], []
    with torch.no_grad():
        for ids, lengths, labels in batches:
            out.append(model(ids, lengths).argmax(1))
            gold.append(labels)
    return torch.cat(out).numpy(), torch.cat(gold).numpy()

print(f"{len(NAMES)} classes, so random guessing is {1 / len(NAMES):.3f}\n")

for kind in ("rnn", "lstm"):
    torch.manual_seed(SEED)
    model = Recurrent(len(vocab), len(NAMES), kind)
    loss_fn, optimiser = nn.CrossEntropyLoss(), torch.optim.Adam(model.parameters(), lr=1e-3)
    print(f"{kind.upper()}  {sum(p.numel() for p in model.parameters()):,} parameters")
    train_batches = shuffled_batches()
    for epoch in range(1, 6):
        model.train()
        started, running = time.time(), 0.0
        for ids, lengths, labels in train_batches:
            optimiser.zero_grad()
            loss = loss_fn(model(ids, lengths), labels)
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), 5.0)
            optimiser.step()
            running += loss.item() * len(labels)
        print(f"  epoch {epoch}  loss={running / len(train_df):.4f}"
              f"  test={accuracy(model, test_batches):.3f}  ({time.time() - started:.0f}s)")
    print()
    if kind == "lstm":
        predicted, gold = predict(model, test_batches)
        print(classification_report(gold, predicted, target_names=NAMES,
                                    digits=3, zero_division=0))
        print("confusion matrix (rows = true, columns = predicted)")
        print(confusion_matrix(gold, predicted))
        off = abs(predicted - gold)
        print(f"\nexactly right      : {(off == 0).mean():.3f}")
        print(f"within one star    : {(off <= 1).mean():.3f}")
        print(f"three or more off  : {(off >= 3).mean():.3f}")

Дәрісте

Білімді тексеру

Аяқтар алдында бірге жауап беретін бес сұрақ.

1-сұрақ

RNN сөз векторларының орташасы бере алмайтын нені бере алады?

Жауапты көрсету

Тәртіпті. жақсы емес тіркесінен кейінгі жасырын күй емес жақсы тіркесінен кейінгіден өзгеше; ал орташа екеуі үшін бірдей, беттің басында өлшенгендей.

2-сұрақ

Жасырын күй өлшемі 128 болса, rnn.weight_ih_l0 неге (512, 100) пішінді?

Жауапты көрсету

512 = 4 × 128. LSTM әр қадамда төрт нәрсе есептейді — forget, input және output қақпалары мен кандидат мәні, — ал PyTorch олардың салмақ матрицаларын біреуіне жинайды.

3-сұрақ

LSTM-нің оқыту шығыны түсе береді, ал тест дәлдігі орнында тұр. Не болып жатыр және нені өзгертер едіңіз?

Жауапты көрсету

Қайта оқыту — 8 000 қысқа пікірге қарсы 700 000 параметр. Нұсқалар: ертерек тоқтау, dropout-ты көтеру, модельді кішірейту немесе көбірек дерек беру. Соңғысын өлшедік: 40 000 пікір оны 0.772-ден 0.803-ке көтерді.

4-сұрақ

4-дәрістегі word2vec-тен бастау неге бірінші эпохада сонша көмектеседі?

Жауапты көрсету

Параметрлердің 83%-ы — эмбеддингтер кестесі, әрі оның 5 916 жолының 5 743-і қандай сөздер бірдей ұстанатынын біліп келеді. Желі шудан емес, құрылымнан бастайды: бір эпохадан кейін 0.727-ге қарсы 0.785.

5-сұрақ

score моделінің дәлдігі 0.374. Неге бұл хабарлауға ең ақпаратты сан емес?

Жауапты көрсету

Кластар реттелген, ал accuracy әр қатені бірдей санайды. Сол модель жағдайлардың 70,4%-ында бір жұлдыз шегінде, ал үш және одан көп қателік небәрі 12,2% — оны қолдану-қолданбауды шешетін адамға дәл осы керек.

Дәрістен кейін

Практикалық тапсырмалар

lesson5.py тұрған қалтада жұмыс істеңіз. Әсерді емес, өлшемді хабарлаңыз.

1-тапсырма — Жасырын күй өлшемі

polarity-де --hidden 32, 128 және 512 іске қосыңыз. Үлкенірек жасырын күй көмектесе ме?

Жауапты көрсету

Ол тез көмектеспей қалады да, кейін зиян етеді деп күтіңіз: сол 8 000 пікірде сыйымдылық көбейсе, қайта оқыту жылдамдайды. Тек соңғы санға емес, оқыту шығыны мен тест дәлдігі арасындағы алшақтыққа қараңыз.

2-тапсырма — Екі бағытты

LSTM-ге bidirectional=True қосыңыз. Жасырын күй енді екі вектор екенін ескеріңіз — сызықтық қабатқа не бересіз?

Жауапты көрсету

Соңғы алға және кері күйлерді torch.cat([hidden[-2], hidden[-1]], dim=1) арқылы біріктіріп, сызықтық қабатты 2 * hidden енді етіңіз. Тоғыз сөзді пікірлерде ұтыс әдетте шағын; қаншалықты екенін айтыңыз.

3-тапсырма — Эмбеддингтерді қатыру

--pretrained кезінде embedding.weight.requires_grad = False қойыңыз. Жақсы ма, нашар ма және неге?

Жауапты көрсету

Қатыру параметрлердің 83%-ын оқытудан алып тастайды, сондықтан қайта оқыту әлдеқайда аз, бірақ векторларды сентиментке бейімдеу мүмкін емес. Шағын оқыту жинағында қатырылған нұсқа жиі ұтады; 40 000 пікірде әдетте ұтылады. Екеуін де өлшеңіз.

4-тапсырма — max_len

Пікірлер 60 сөзге қиылады. Бұл шынымен қаншасына тиеді және --max-len 20 кезінде не болады?

Жауапты көрсету

Орташа пікір — шамамен тоғыз token, ең ұзыны — 271, сондықтан 60 өте азын ғана қияды. Бұл — өз алдына нәтиже: бұл датасет рекурренттік модельге жұмыс істейтін ұзындықты дерлік бермейді.

5-тапсырма — Packing-ті алып тастау

Толтырылған батчты pack_padded_sequence-сіз тікелей LSTM-ге беріңіз. Дәлдікке не болады?

Жауапты көрсету

Сонда қысқа пікірдің соңғы жасырын күйі <pad> қадамдарынан жиналады. Төмендеуді күтіңіз, әрі ол ең қысқа пікірлерде ең күшті болады — тест жинағын ұзындығы бойынша бөліп тексеріңіз.

6-тапсырма — Класс салмақтары

score моделін теңгерілмеген деректе CrossEntropyLoss(weight=…) арқылы оқытыңыз. Орта кластар қалай жылжиды?

Жауапты көрсету

Салмақтар 19 600-дің орнына 140 126 пікірдің бәрін қолдануға мүмкіндік береді. Тек accuracy-ге емес, шатасу матрицасының үш жұлдызды жолына және бір жұлдыз шегіндегі санға қараңыз.

7-тапсырма — Бағаны сан ретінде қарау

5 класты шығысты бір санмен және MSELoss-пен ауыстырып, содан дөңгелектеңіз. Реттелген құрылым көмектесе ме?

Жауапты көрсету

Регрессия 4-тің 3 пен 5-тің арасында екенін біледі, ал жіктеу білмейді. Екі метрика бойынша салыстырыңыз: дәл түсу жиі төмендейді, ал орташа абсолюттік қате жақсарады.

8-тапсырма — Ерте тоқтату

KazSAnDRA беретін валидациялық бөлуді қолданып, соңғы эпоханың орнына ең жақсысын алыңыз. Бұл нәтижені қаншалықты өзгертеді?

Жауапты көрсету

Біздің жүгірістер оқытудың ортасында шыңға жетеді — 2-эпохада 0.799, 1-эпохада 0.803 — содан кейін төмендейді. Эпоханы тест жинағы бойынша таңдау — алдау болар еді; валидациялық бөлу дәл сол үшін бар.

9-тапсырма — word2vec орнына fastText

Эмбеддинг қабатын models/fasttext.model файлынан инициализациялап, сирек сөздерді қоса әр жолды толтырыңыз. Қосымша қамту өтеле ме?

Жауапты көрсету

fastText 5 743 емес, 5 916 жолдың бәріне вектор бере алады. Қосылатын жолдар — сирек сөздер, сондықтан ұтыс шағын болуы мүмкін — дәлдіктің қасында толтырылған жол санын хабарлаңыз.

10-тапсырма — Seed-тер

LSTM-ді бес түрлі seed-пен іске қосыңыз. Шашырау қандай және ол осы беттегі қандай да бір қорытындыны өзгерте ме?

Жауапты көрсету

Бұл — мұндағы ең маңызды тапсырма. Шашырауды біз мәнді деп атаған айырмалармен салыстырыңыз: RNN мен LSTM арасы 0.017, word2vec-тен бастау 0.015. Егер seed шашырауы сондай болса, бұл тұжырымдардың артында бір жүгірістен көбі тұруы керек.

Өзін-өзі тексеру тесті

Бірден кері байланыс беретін он сұрақ.

  1. RNN-нің жасырын күйі дегеніміз не?

    Ол нөлден басталып, сол салмақтармен әр сөзге бір рет жаңарады.

  2. Неге RNN «жақсы емес» пен «емес жақсы» тіркестерін ажыратады, ал TF-IDF ажыратпайды?

    Санау да, орташалау да өз құрылысы бойынша тәртіпті көрмейді.

  3. LSTM-нің ұяшық күйі қандай мәселені шешеді?

    Ол қайталанатын матрицаға көбейту емес, қосу жолымен жүреді.

  4. forget қақпасы не істейді?

    1-ге жақын мәндер жадты көп сөз бойы сақтайды; 0-ге жақындары оны өшіреді.

  5. Батчқа неге толтыру мен ұзындықтар керек?

    pack_padded_sequence 15 ұяшықтың бәрінің орнына 9 нағыз қадам орындады.

  6. Параметрлердің көбі қай қабатта?

    Сондықтан оны жақсы инициализациялау біз сынағанның бәрінен маңыздырақ болды.

  7. Оқыту шығыны түседі, тест дәлдігі түспейді. Бұл…

    Модель әлі де үйреніп жатыр — нақтырақ айтқанда, оқыту жинағын.

  8. Эмбеддинг қабатын 4-дәрістің word2vec-інен бастау ең көп көмектесті…

    Ол шудан емес, құрылымнан бастайды, сондықтан оған әлдеқайда аз өту керек.

  9. score моделі 0.374 берді, ал болжау 0.200 береді. Мұны хабарлаудың ең жақсы жолы…

    Реттелген шкалада «қаншалықты қате» «қаншалықты жиі» дегеннен кем маңызды емес.

  10. Рекурренттік желіні неге сөздер бойынша параллельдеу мүмкін емес?

    Келесі дәрісте attention дәл осы шектеуді алып тастайды.

Қорытынды

Бүгін біз үйрендік

  • Тәртіп сөздер қабына неге түсетінін — өлшеумен
  • Жасырын күйді және орындарға ортақ салмақтарды

  • LSTM-нің ұяшық күйін және оның үш қақпасын
  • Сөздікті, <pad>, <unk> және packing-ті
  • PyTorch-тағы модель класы мен оқыту циклін
