import re
from collections import Counter

import numpy as np
from gensim.models import Word2Vec

word2vec = Word2Vec.load("models/word2vec.model")

a = "қосымша өте жақсы емес"      # "the app is not very good"
b = "емес жақсы өте қосымша"      # the same four words, shuffled

def bag_of_words(text):
    return Counter(re.findall(r"\w+", text.lower()))

def document_vector(text):
    words = re.findall(r"\w+", text.lower())
    return np.mean([word2vec.wv[w] for w in words if w in word2vec.wv], axis=0)

print("«" + b + "»")

Теория · Модель

Рекурренттік желі дегеніміз не

Әр сөзге бір рет қолданылатын, әрі осы уақытқа дейін оқығанының бәрінің қорытындысын өзімен алып жүретін бір шағын желі.

Рекурренттік нейрондық желі жасырын күй сақтайды: нөлден басталып, әр сөзде қайта жазылатын вектор, h = f(h, сөз). Әр орында сол бір f және сол бір салмақтар жұмыс істейді. Соңғы сөзден кейін осы жалғыз вектор — модельдің пікір туралы қорытындысы.

Төрт желі емес, бір желі төрт рет салынған. Оның салмақтары барлық орындарға ортақ, сондықтан үш сөзді пікір де, қырық сөзді пікір де дәл сол параметрлер арқылы өтеді.

Бір қадамның ішінде

Бәрі бір жол арифметикаға сыяды. Қадам екі нәрсені алады — алдыңғы сөзден қалған күй мен ағымдағы сөзді, — әрқайсысын өз салмақ матрицасына көбейтіп, қосады да, нәтижесін қысады.

Қарапайым RNN-нің бір қадамы. W сөзге, U алдыңғы күйге қолданылады; олардың қосындысын tanh жаңа күйге қысады.

hₜ = tanh(W·xₜ + U·hₜ₋₁ + b) оқимыз
xₜ
ағымдағы сөз эмбеддинг векторы түрінде — біздің модельде 100 сан
hₜ₋₁
алдыңғы сөзден кейін желі білгені — 128 сан
W, U
екі салмақ матрицасы. Олар әр орында бірдей: «рекурренттік» дегеннің мәні осы
tanh
әр санды (−1, 1) аралығына қысады, күйдің шексіз өсуін тоқтатады
hₜ
жаңа күй, ол келесі сөзде hₜ₋₁ болады

Оқылуы: жады басқа еш жерде жоқ. Желінің бірінші сөз туралы есте сақтағысы келгенінің бәрі одан кейінгі барлық қайта жазудан h ішінде аман өтуі керек.

Күйдің қалай берілетінін көреміз

Ең түсініктісі — рекурренттілікті қолмен жүгіртіп, жауапты PyTorch-пен салыстыру. Төменде төрт өлшемді күй мен төрт сөз: әр сан бетке сыятындай шағын. 

import torch
import torch.nn as nn

torch.manual_seed(0)

# A tiny RNN so the numbers fit on the page: 3-dimensional words, 4-dimensional state.
rnn = nn.RNN(input_size=3, hidden_size=4, batch_first=True)

words = torch.tensor([[[1.0, 0.0, 0.0],    # қосымша
                       [0.0, 1.0, 0.0],    # өте
                       [0.0, 0.0, 1.0],    # жақсы
                       [1.0, 1.0, 0.0]]])  # емес

# --- what nn.RNN does, written out ----------------------------------------
W_ih, W_hh = rnn.weight_ih_l0, rnn.weight_hh_l0
b_ih, b_hh = rnn.bias_ih_l0, rnn.bias_hh_l0

h = torch.zeros(4)                                      # the state starts at zero
print("h0 (before any word):", h.numpy().round(3))
for step in range(4):
    x = words[0, step]
    h = torch.tanh(W_ih @ x + b_ih + W_hh @ h + b_hh)   # <- the whole recurrence
    print(f"h{step + 1} after word {step + 1}   :", h.detach().numpy().round(3))

# --- the same thing, done by PyTorch --------------------------------------
outputs, last = rnn(words)
print("\nPyTorch's last hidden state:", last[0, 0].detach().numpy().round(3))
print("matches our loop           :", bool(torch.allclose(h, last[0, 0], atol=1e-6)))
print("\noutputs holds every step   :", tuple(outputs.shape), "= (batch, words, hidden)")
print("its last row is the last hidden state:",
      bool(torch.allclose(outputs[0, -1], last[0, 0])))

Мұнда көрінетін үш нәрсе. Біріншісі: h₀ — тұтас нөл, желі білімсіз бастап, бәрін сөздерден құрады. Екіншісі: әр баған тек алдыңғы бағаннан және ағымдағы сөзден есептеледі — тәртіптің маңызды болатыны да сондықтан: сөздерді басқа ретпен берсеңіз, бірінші бағаннан кейінгінің бәрі өзгереді. Үшіншісі: nn.RNN — біз жазған дәл сол цикл; фреймворк басқа алгоритм емес, жылдам іске асыру береді.

Сөздер ретімен келетіндіктен, жақсы емес пен емес жақсы енді әртүрлі жасырын күй береді. Қосымша механизмнің бар себебі осы: курстағы басқа ештеңе бұл екеуін ажырата алмайтын.

Теория · Түзету

Қарапайым RNN неге ұмытады және LSTM не істейді

Жасырын күй әр сөзде қайта жазылады. Оны қырық рет қайта жазыңыз — басында ішінде не болғаны жоқ болады.

Градиенттің өшуі, сөзбен

Оқыту кері жүреді. Қате пікірдің соңында есептеледі, ал бірінші сөз маңызды болғанын анықтау үшін сигнал арадағы барлық қадам арқылы кері өтуі керек. Әр қадамда ол сол бір екі нәрсеге көбейтіледі: ешқашан 1-ден аспайтын tanh туындысына және рекурренттік U матрицасына. Шамамен бірдей көбейткішке қырық рет көбейтіңіз — екінің бірі болады.

Көбейткіш 1-ден кіші — өшу

0.540 шамамен 10−12. Сигнал нөл болып жетеді, ерте салмақтар қозғалмайды, ал модель пікірдің басы болғанын мүлде білмей қалады.

Көбейткіш 1-ден үлкен — жарылыс

1.540 шамамен 107. Бір батч орасан жаңарту беріп, салмақтар мағынасыздыққа ұшады. Бұл жартысын түзету оңай, ал оқыту циклындағы clip_grad_norm_ — дәл сол жамау.

Градиенттің өшуі, өлшенген

Бұл әңгіме емес — мұны басып шығаруға болады. 40 сөзден тұратын тізбекті береміз, соңғы күйден кері бір сигнал жібереміз де, оның әр кіріс орнына қаншасы жететінін сұраймыз. 

import torch
import torch.nn as nn

LENGTH, DIM, HIDDEN = 40, 32, 64

def gradient_by_position(cell):
    """How much does the final state still depend on the word at each position?"""
    torch.manual_seed(0)
    words = torch.randn(1, LENGTH, DIM, requires_grad=True)
    _, state = cell(words)
    hidden = state[0] if isinstance(state, tuple) else state   # LSTM returns (h, c)
    hidden.sum().backward()                    # one signal, sent back from the end
    return words.grad[0].norm(dim=1)           # one number per position

torch.manual_seed(0)
rnn = nn.RNN(DIM, HIDDEN, batch_first=True)
torch.manual_seed(0)
lstm = nn.LSTM(DIM, HIDDEN, batch_first=True)

# The standard trick: start the forget gate open. nn.LSTM stacks the gates as
# (input, forget, cell, output), so the forget bias is the second quarter.
torch.manual_seed(0)
open_lstm = nn.LSTM(DIM, HIDDEN, batch_first=True)
with torch.no_grad():
    open_lstm.bias_ih_l0[HIDDEN:2 * HIDDEN].fill_(3.0)      # sigmoid(3) ≈ 0.95

grads = {name: gradient_by_position(cell) for name, cell in
         (("RNN", rnn), ("LSTM", lstm), ("LSTM, forget open", open_lstm))}

print(f"{'steps back':>10} {'RNN':>10} {'LSTM':>10} {'LSTM, forget open':>19}")
for back in (0, 1, 2, 5, 10, 20, 30, 39):
    i = LENGTH - 1 - back
    row = "  ".join(f"{g[i]:>8.1e}" for g in grads.values())
    print(f"{back:>10}   {row}")

print("\nhow much of the last word's signal still reaches the first word:")
for name, g in grads.items():
    ratio = float(g[0] / g[-1])
    per_step = ratio ** (1 / (LENGTH - 1))
    print(f"  {name:<18} {ratio:>9.1e}   ×{per_step:.2f} per step")

print("\nThe RNN loses half the signal at every step, so after 39 there is nothing")
print("left. The LSTM's default init is barely better. But hold the forget gate")
print("open and the factor becomes 1.00 — the signal survives the whole sequence.")
print("That knob is the entire point of the cell state, and an LSTM can learn it.")

Мәні: LSTM градиенттің өшуін жоймайды. Ол модельге оны өшіруге болатын тетік береді әрі оны қашан бұрау керегін — әр өлшем мен әр пікір үшін — оқытудың өзі шешеді. Қарапайым RNN-де ондай тетік жоқ.

LSTM нені қосады

LSTM екінші вектор — ұяшық күйін қосады, ол қадамнан қадамға дерлік тиіссіз өтеді: оны көбейтеді және оған қосады, бірақ ешқашан салмақ матрицасы арқылы өткізбейді. Үш қақпа, әрқайсысы 0 мен 1 арасында сан беретін шағын желі, нені өшіру, нені қосу және нені көрсету керегін шешеді.

Көк сызық — ұяшық күйі. Оны солдан оңға қарай қадағалаңыз: жолында бір көбейту мен бір қосу кездеседі, басқа ештеңе жоқ. Градиент дәл осы тік жолмен кері жүреді.

Екі теңдеу, мүшелеп
f ⊙ cₜ₋₁
ұмыту. 1-ге жуық мән жадының сол өлшемін сақтайды, 0-ге жуығы өшіреді. Бұл — жоғарыда өлшенген тетік
i ⊙ ĉ
кіріс. ĉ — жаңа үміткер мән, ал i оның қаншасын жазуды шешеді
cₜ = f ⊙ cₜ₋₁ + i ⊙ ĉ
бүкіл жаңарту: болғанын масштабта, жаңасын қос. Қайта жазу емес, қосу
hₜ = o ⊙ tanh(cₜ)
шығыс. Жады мен көрінетін күй енді екі бөлек нәрсе — ұяшық бірдеңені көрсетпей-ақ сақтай алады

RNN-мен салыстырыңыз: онда h әрі жады, әрі шығыс болатын, әрі әр қадам оны қайта жазатын. Мұнда жадының өз қорғалған жолағы бар.

Қақпалар, басып шығарылған

Бұрынғыдай тәсіл: теңдеулерді қолмен жазып, әр қадамдағы қақпа мәндерін басып шығару және nn.LSTM-мен салыстыру. 

import torch
import torch.nn as nn

torch.manual_seed(0)

lstm = nn.LSTM(input_size=3, hidden_size=4, batch_first=True)
words = torch.randn(1, 6, 3)

# nn.LSTM stacks the four gates into one matrix, in this order:
i_w, f_w, g_w, o_w = lstm.weight_ih_l0.chunk(4)
i_u, f_u, g_u, o_u = lstm.weight_hh_l0.chunk(4)
i_b, f_b, g_b, o_b = (lstm.bias_ih_l0 + lstm.bias_hh_l0).chunk(4)

h = torch.zeros(4)
c = torch.zeros(4)
print(f"{'step':>4}  {'forget':>22}  {'input':>22}  {'cell state c':>26}")
for t in range(6):
    x = words[0, t]
    f = torch.sigmoid(f_w @ x + f_u @ h + f_b)      # what to keep from c
    i = torch.sigmoid(i_w @ x + i_u @ h + i_b)      # how much of the new value to add
    g = torch.tanh(g_w @ x + g_u @ h + g_b)         # the candidate value
    o = torch.sigmoid(o_w @ x + o_u @ h + o_b)      # what to reveal
    c = f * c + i * g                               # <- the highway: multiply, then add
    h = o * torch.tanh(c)                           # <- what the next layer sees
    print(f"{t + 1:>4}  {str(f.detach().numpy().round(2)):>22}"
          f"  {str(i.detach().numpy().round(2)):>22}"
          f"  {str(c.detach().numpy().round(2)):>26}")

_, (h_torch, c_torch) = lstm(words)
print("\nmatches nn.LSTM — h:", bool(torch.allclose(h, h_torch[0, 0], atol=1e-6)),
      " c:", bool(torch.allclose(c, c_torch[0, 0], atol=1e-6)))

Практика · Модель

PyTorch-тағы желі

Үш қабат және он бір жол. RNN мен LSTM арасында ауысу — бір сөз.

Әр қабаттағы деректің пішіні

Класқа дейін — ол іске асыратын конвейер: төрт қабат және олардың арасында ағып өтетін нәрсенің пішіні.

Үш пікірден тұратын батч бүкіл желі арқылы өтеді. Көк блоктарда салмақтар, сұрғылттарында олардың арасындағы деректер.

PAD = 0

class Recurrent(nn.Module):
    """Embedding -> RNN or LSTM -> Linear."""

    def __init__(self, vocab_size, classes, kind="lstm", dim=100, hidden=128, dropout=0.3):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, dim, padding_idx=PAD)
        cell = nn.LSTM if kind == "lstm" else nn.RNN
        self.rnn = cell(dim, hidden, batch_first=True)
        self.dropout = nn.Dropout(dropout)
        self.out = nn.Linear(hidden, classes)

    def forward(self, ids, lengths):
        embedded = self.dropout(self.embedding(ids))          # (batch, words, dim)
        packed = nn.utils.rnn.pack_padded_sequence(
            embedded, lengths, batch_first=True, enforce_sorted=False)
        _, state = self.rnn(packed)
        hidden = state[0] if isinstance(state, tuple) else state   # LSTM returns (h, c)
        return self.out(self.dropout(hidden[-1]))             # (batch, classes)

Дәрісте

Білімді тексеру

Аяқтар алдында бірге жауап беретін бес сұрақ.

1-сұрақ

RNN сөз векторларының орташасы бере алмайтын нені бере алады?

Жауапты көрсету

Тәртіпті. жақсы емес тіркесінен кейінгі жасырын күй емес жақсы тіркесінен кейінгіден өзгеше; ал орташа екеуі үшін бірдей, беттің басында өлшенгендей.

2-сұрақ

Жасырын күй өлшемі 128 болса, rnn.weight_ih_l0 неге (512, 100) пішінді?

Жауапты көрсету

512 = 4 × 128. LSTM әр қадамда төрт нәрсе есептейді — forget, input және output қақпалары мен кандидат мәні, — ал PyTorch олардың салмақ матрицаларын біреуіне жинайды.

3-сұрақ

LSTM-нің оқыту шығыны түсе береді, ал тест дәлдігі орнында тұр. Не болып жатыр және нені өзгертер едіңіз?

Жауапты көрсету

Қайта оқыту — 8 000 қысқа пікірге қарсы 700 000 параметр. Нұсқалар: ертерек тоқтау, dropout-ты көтеру, модельді кішірейту немесе көбірек дерек беру. Соңғысын өлшедік: 40 000 пікір оны 0.772-ден 0.803-ке көтерді.

4-сұрақ

4-дәрістегі word2vec-тен бастау неге бірінші эпохада сонша көмектеседі?

Жауапты көрсету

Параметрлердің 83%-ы — эмбеддингтер кестесі, әрі оның 5 916 жолының 5 743-і қандай сөздер бірдей ұстанатынын біліп келеді. Желі шудан емес, құрылымнан бастайды: бір эпохадан кейін 0.727-ге қарсы 0.785.

5-сұрақ

score моделінің дәлдігі 0.374. Неге бұл хабарлауға ең ақпаратты сан емес?

Жауапты көрсету

Кластар реттелген, ал accuracy әр қатені бірдей санайды. Сол модель жағдайлардың 70,4%-ында бір жұлдыз шегінде, ал үш және одан көп қателік небәрі 12,2% — оны қолдану-қолданбауды шешетін адамға дәл осы керек.

Дәрістен кейін

Практикалық тапсырмалар

lesson5.py тұрған қалтада жұмыс істеңіз. Әсерді емес, өлшемді хабарлаңыз.

1-тапсырма — Жасырын күй өлшемі

polarity-де --hidden 32, 128 және 512 іске қосыңыз. Үлкенірек жасырын күй көмектесе ме?

Жауапты көрсету

Ол тез көмектеспей қалады да, кейін зиян етеді деп күтіңіз: сол 8 000 пікірде сыйымдылық көбейсе, қайта оқыту жылдамдайды. Тек соңғы санға емес, оқыту шығыны мен тест дәлдігі арасындағы алшақтыққа қараңыз.

2-тапсырма — Екі бағытты

LSTM-ге bidirectional=True қосыңыз. Жасырын күй енді екі вектор екенін ескеріңіз — сызықтық қабатқа не бересіз?

Жауапты көрсету

Соңғы алға және кері күйлерді torch.cat([hidden[-2], hidden[-1]], dim=1) арқылы біріктіріп, сызықтық қабатты 2 * hidden енді етіңіз. Тоғыз сөзді пікірлерде ұтыс әдетте шағын; қаншалықты екенін айтыңыз.

3-тапсырма — Эмбеддингтерді қатыру

--pretrained кезінде embedding.weight.requires_grad = False қойыңыз. Жақсы ма, нашар ма және неге?

Жауапты көрсету

Қатыру параметрлердің 83%-ын оқытудан алып тастайды, сондықтан қайта оқыту әлдеқайда аз, бірақ векторларды сентиментке бейімдеу мүмкін емес. Шағын оқыту жинағында қатырылған нұсқа жиі ұтады; 40 000 пікірде әдетте ұтылады. Екеуін де өлшеңіз.

4-тапсырма — max_len

Пікірлер 60 сөзге қиылады. Бұл шынымен қаншасына тиеді және --max-len 20 кезінде не болады?

Жауапты көрсету

Орташа пікір — шамамен тоғыз token, ең ұзыны — 271, сондықтан 60 өте азын ғана қияды. Бұл — өз алдына нәтиже: бұл датасет рекурренттік модельге жұмыс істейтін ұзындықты дерлік бермейді.

5-тапсырма — Packing-ті алып тастау

Толтырылған батчты pack_padded_sequence-сіз тікелей LSTM-ге беріңіз. Дәлдікке не болады?

Жауапты көрсету

Сонда қысқа пікірдің соңғы жасырын күйі <pad> қадамдарынан жиналады. Төмендеуді күтіңіз, әрі ол ең қысқа пікірлерде ең күшті болады — тест жинағын ұзындығы бойынша бөліп тексеріңіз.

6-тапсырма — Класс салмақтары

score моделін теңгерілмеген деректе CrossEntropyLoss(weight=…) арқылы оқытыңыз. Орта кластар қалай жылжиды?

Жауапты көрсету

Салмақтар 19 600-дің орнына 140 126 пікірдің бәрін қолдануға мүмкіндік береді. Тек accuracy-ге емес, шатасу матрицасының үш жұлдызды жолына және бір жұлдыз шегіндегі санға қараңыз.

7-тапсырма — Бағаны сан ретінде қарау

5 класты шығысты бір санмен және MSELoss-пен ауыстырып, содан дөңгелектеңіз. Реттелген құрылым көмектесе ме?

Жауапты көрсету

Регрессия 4-тің 3 пен 5-тің арасында екенін біледі, ал жіктеу білмейді. Екі метрика бойынша салыстырыңыз: дәл түсу жиі төмендейді, ал орташа абсолюттік қате жақсарады.

8-тапсырма — Ерте тоқтату

KazSAnDRA беретін валидациялық бөлуді қолданып, соңғы эпоханың орнына ең жақсысын алыңыз. Бұл нәтижені қаншалықты өзгертеді?

Жауапты көрсету

Біздің жүгірістер оқытудың ортасында шыңға жетеді — 2-эпохада 0.799, 1-эпохада 0.803 — содан кейін төмендейді. Эпоханы тест жинағы бойынша таңдау — алдау болар еді; валидациялық бөлу дәл сол үшін бар.

9-тапсырма — word2vec орнына fastText

Эмбеддинг қабатын models/fasttext.model файлынан инициализациялап, сирек сөздерді қоса әр жолды толтырыңыз. Қосымша қамту өтеле ме?

Жауапты көрсету

fastText 5 743 емес, 5 916 жолдың бәріне вектор бере алады. Қосылатын жолдар — сирек сөздер, сондықтан ұтыс шағын болуы мүмкін — дәлдіктің қасында толтырылған жол санын хабарлаңыз.

10-тапсырма — Seed-тер

LSTM-ді бес түрлі seed-пен іске қосыңыз. Шашырау қандай және ол осы беттегі қандай да бір қорытындыны өзгерте ме?

Жауапты көрсету

Бұл — мұндағы ең маңызды тапсырма. Шашырауды біз мәнді деп атаған айырмалармен салыстырыңыз: RNN мен LSTM арасы 0.017, word2vec-тен бастау 0.015. Егер seed шашырауы сондай болса, бұл тұжырымдардың артында бір жүгірістен көбі тұруы керек.

Өзін-өзі тексеру тесті

Бірден кері байланыс беретін он сұрақ.

  1. RNN-нің жасырын күйі дегеніміз не?

    Ол нөлден басталып, сол салмақтармен әр сөзге бір рет жаңарады.

  2. Неге RNN «жақсы емес» пен «емес жақсы» тіркестерін ажыратады, ал TF-IDF ажыратпайды?

    Санау да, орташалау да өз құрылысы бойынша тәртіпті көрмейді.

  3. LSTM-нің ұяшық күйі қандай мәселені шешеді?

    Ол қайталанатын матрицаға көбейту емес, қосу жолымен жүреді.

  4. forget қақпасы не істейді?

    1-ге жақын мәндер жадты көп сөз бойы сақтайды; 0-ге жақындары оны өшіреді.

  5. Батчқа неге толтыру мен ұзындықтар керек?

    pack_padded_sequence 15 ұяшықтың бәрінің орнына 9 нағыз қадам орындады.

  6. Параметрлердің көбі қай қабатта?

    Сондықтан оны жақсы инициализациялау біз сынағанның бәрінен маңыздырақ болды.

  7. Оқыту шығыны түседі, тест дәлдігі түспейді. Бұл…

    Модель әлі де үйреніп жатыр — нақтырақ айтқанда, оқыту жинағын.

  8. Эмбеддинг қабатын 4-дәрістің word2vec-інен бастау ең көп көмектесті…

    Ол шудан емес, құрылымнан бастайды, сондықтан оған әлдеқайда аз өту керек.

  9. score моделі 0.374 берді, ал болжау 0.200 береді. Мұны хабарлаудың ең жақсы жолы…

    Реттелген шкалада «қаншалықты қате» «қаншалықты жиі» дегеннен кем маңызды емес.

  10. Рекурренттік желіні неге сөздер бойынша параллельдеу мүмкін емес?

    Келесі дәрісте attention дәл осы шектеуді алып тастайды.
